Required Environment Variables (.env.local):

  # Admin Credentials
  ADMIN_EMAIL=admin@auapw.com
  ADMIN_PASSWORD=AUAPWAdmin123!

  # Database
  DATABASE_URL=postgresql://user:password@host:5432/auapw

  # Email Service (Resend)
  RESEND_API_KEY=your_resend_api_key_here

  # Microsoft Teams (Internal Notifications)
  TEAMS_WEBHOOK_URL=https://outlook.webhook.office.com/...

  # Payment Gateway (Stripe)
  STRIPE_SECRET_KEY=sk_test_...
  STRIPE_PUBLIC_KEY=pk_test_...

  # Fraud Detection
  SIGNIFYD_API_KEY=your_signifyd_key
  SIFT_API_KEY=your_sift_key

  # CRM Integration
  HUBSPOT_API_KEY=your_hubspot_key
